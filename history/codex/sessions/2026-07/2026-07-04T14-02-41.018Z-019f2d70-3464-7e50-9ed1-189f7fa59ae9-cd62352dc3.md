    torch::Tensor linear_hxk(const torch::Tensor& x, const torch::Tensor& weight_hxk, const torch::Tensor& bias) const {
        return at::linear(x, weight_hxk.transpose(0, 1), bias);

    torch::Tensor build_tokens(const torch::Tensor& state_u8) const {
        const torch::Tensor states = state_u8.index({torch::indexing::Slice(), torch::indexing::Slice(0, state_len)})
                                         .to(torch::kLong);

        torch::Tensor pieces = fast_piece_static.unsqueeze(0).expand({batch, -1, -1}).clone();

            torch::Tensor positions = piece_positions.index({torch::indexing::Slice(), static_cast<std::int64_t>(slot)});
            torch::Tensor state_values = states.index_select(1, positions).reshape({-1});
            torch::Tensor gathered = fast_slot_projected.index({static_cast<std::int64_t>(slot)})
                                         .index_select(0, state_values)
                                         .view({batch, static_cast<std::int64_t>(num_pieces), static_cast<std::int64_t>(d_model)});
            torch::Tensor mask = piece_mask.index({torch::indexing::Slice(), static_cast<std::int64_t>(slot)})
                                     .to(dtype)
                                     .view({1, static_cast<std::int64_t>(num_pieces), 1});
            pieces = pieces + gathered * mask;
        }
        torch::Tensor cls = cls_token.view({1, 1, static_cast<std::int64_t>(d_model)}).expand({batch, 1, -1});
        return torch::cat({cls, pieces}, 1);
    }

        for (const TransformerBlock& block : blocks) {
            torch::Tensor y = layer_norm(x, block.ln1_gamma, block.ln1_beta);
            torch::Tensor qkv = linear_hxk(y, block.qkv_weight, block.qkv_bias);
            qkv = qkv.view({batch,
                            static_cast<std::int64_t>(seq_len),
                            3,
                            static_cast<std::int64_t>(nhead),
                            static_cast<std::int64_t>(head_dim)});
            torch::Tensor q = qkv.select(2, 0).permute({0, 2, 1, 3});
            torch::Tensor k = qkv.select(2, 1).permute({0, 2, 1, 3});
            torch::Tensor v = qkv.select(2, 2).permute({0, 2, 1, 3});
            torch::Tensor attn = at::scaled_dot_product_attention(q, k, v, std::nullopt, 0.0, false, std::nullopt, false);
            torch::Tensor context = attn.permute({0, 2, 1, 3})
                                        .reshape({batch,
                                                  static_cast<std::int64_t>(seq_len),
                                                  static_cast<std::int64_t>(d_model)});
            x = x + linear_hxk(context, block.attn_out_weight, block.attn_out_bias);

            y = layer_norm(x, block.ln2_gamma, block.ln2_beta);
            y = linear_hxk(y, block.ff1_weight, block.ff1_bias);
            y = y * torch::sigmoid(y);
            x = x + linear_hxk(y, block.ff2_weight, block.ff2_bias);
        }
        torch::Tensor cls = layer_norm(x.index({torch::indexing::Slice(), 0, torch::indexing::Slice()}), output_ln_gamma, output_ln_beta);
        return linear_hxk(cls, output_weight, output_bias);
    }

PyTorch can put capture-time allocations in a private pool, but correctness depends on retaining tensors and stable execution. This needs a microtest plus Nsight proof before production use.

**Profiling Plan**

Nsight Systems first:

```bash
nsys profile --trace=cuda,nvtx,osrt,cublas --stats=true --force-overwrite=true -o libtorch_s1_graph ./stream1_transformer_libtorch_benchmark --weight-dir ... --batches 384 --warmup 10 --iters 200 --device cuda:0 --cuda-graph
```

Success signs: warmup shows eager ATen/cuBLAS/SDPA launches; timed region shows repeated `cudaGraphLaunch`, much fewer CPU-side CUDA API calls per iteration, no `cudaMalloc/cudaFree` or synchronizing `.item()` inside replay window, stable kernel sequence, no unexpected fallback eager launches.

Nsight Compute only after Systems identifies a hot kernel:

```bash
ncu --set full --target-processes all --launch-skip 5 --launch-count 10 --force-overwrite -o s1_sdpa_or_gemm ./stream1_transformer_libtorch_benchmark ...
```

Use it for the dominant SDPA/GEMM/layernorm kernel, not whole-pipeline scheduling. Check tensor-core use, occupancy, memory throughput, and whether SDPA backend is the intended one.

**Main Risks**

- Existing LibTorch `forward()` is allocation-heavy; capture may work but graph memory pool size/lifetime must be controlled.
- SDPA backend can change by device/shape/PyTorch version; pinning/observing backend is mandatory.
- cuBLASLt autotune/cache/workspace behavior must be warmed before capture; no first-use initialization during capture.
- Integrating LibTorch into current production dispatcher would collide architecturally with existing native CUDA graph templates. Keep it benchmark/explicit backend until graph replay is demonstrated with profiler evidence.
- No fallback means failure must be loud: unsupported device, unsupported dtype/backend, capture error, or dynamic shape mismatch should abort.
