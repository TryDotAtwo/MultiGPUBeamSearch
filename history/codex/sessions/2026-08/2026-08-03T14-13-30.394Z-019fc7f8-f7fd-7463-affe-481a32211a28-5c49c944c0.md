    }

    if (!found && solved.stream2_suffix.enabled != 0U) {
        found = stream2_suffix_find_hit(
            solved.stream2_suffix,
            generators,
            parent,
            move,
            central_state,
            zobrist,
            solved.solved_neighborhood,
            use_neighborhood,
            &found_hash,
            &found_suffix_id);
    }

    if (found && solved.solved_count != nullptr) {
        const std::uint32_t idx = atomicAdd(solved.solved_count, 1U);
        if (idx < solved.solved_result_capacity) {
            const std::uint32_t solved_depth =
                solved.current_depth == nullptr ? depth : *solved.current_depth;
            CandidateMeta meta{};
            meta.hash = found_hash;
            meta.parent_idx = parent_idx;
            meta.score_key = GOAL_SCORE_KEY;
            meta.route_packed =
                (static_cast<std::uint32_t>(local_rank) << 16) |
                (static_cast<std::uint32_t>(local_rank) << 8) |
                move;
            solved.solved_meta_list[idx] = meta;
            solved.solved_depth_list[idx] = solved_depth;
            if (solved.solved_suffix_list != nullptr) {
                solved.solved_suffix_list[idx] = found_suffix_id;
            }
        } else {
            atomicExch(solved.solved_overflow, 1U);
        }
        __threadfence_system();
        if (atomicCAS(solved.solved_flag, 0U, 1U) == 0U) {
            if (solved.stop_on_found != 0U) {
                atomicExch(solved.stop_flag, 1U);
            }
        }
    }
}

- Перенести `searchsorted` на TPU через `jnp.searchsorted` можно, но бинарный поиск с нерегулярным доступом не особенно хорошо соответствует TPU.

Для TPU практичнее держать `hashes` и `depths` на устройстве и выполнять полностью JIT-компилируемый lookup:

```python
@jax.jit
def eg_lookup_tpu(states):
    h = jnp.zeros(states.shape[0], dtype=jnp.uint64)
    for i in range(states.shape[1]):
        h ^= eg_ztab[i, states[:, i]]

    pos = jnp.searchsorted(eg_hashes, h)
    valid = pos < eg_hashes.size
    safe_pos = jnp.minimum(pos, eg_hashes.size - 1)

    return jnp.where(
        valid & (eg_hashes[safe_pos] == h),
        eg_depths[safe_pos],
        -1,
    )
```

Но ещё лучше для TPU — dense/direct-address или bucketed lookup с фиксированным числом чтений, если память позволяет. TPU любит статические формы и одинаковую работу для всех элементов; фиксированные 4–8 probes подходят ему лучше бинарного поиска.

Итог: сам перебор в `eg_descend()` не страшен — он выполняется один раз для короткого хвоста. Критично, чтобы массовое пересечение frontier не шло через текущий NumPy `eg_lookup()`. Если оно идёт через отдельный `jnp`-код, нужно сравнивать именно его.
